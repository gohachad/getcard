# getcard.kg — public marketing site.
#
# The order flow and tracking run on this same host (proxied to the app
# container) so a client never changes domain mid-purchase. Those paths are
# private to one customer and must not be crawled: an order page has nothing
# useful to index, and a tracking URL carries a code in the query string.
#
# The pages also carry their own noindex, so this is belt and braces.

User-agent: *
Allow: /
Disallow: /order
Disallow: /track
Disallow: /api/
Disallow: /app-assets/

Sitemap: https://getcard.kg/sitemap.xml
